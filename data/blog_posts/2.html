SQLite3 in Python: Data Oriented & Procedural Programming

Golden Rule: The database is an I/O boundary. Data goes in plain, data comes out plain. Functions transform data. Never wrap rows in objects.

1. The Mindset

In Data Oriented Programming (DOP), we care about:

  • Plain data: dict, tuple, list, namedtuple — never custom classes
  • Functions that transform data: f(data) -> new_data
  • Separating I/O from logic: Query the DB → get plain data → process → write back
# BAD (OOP)
class User:
    def __init__(self, id, name): ...
    def save(self): ...

# GOOD (DOP)
def fetch_user(conn, user_id) -> dict | None: ...
def update_user(conn, user_id, data: dict) -> None: ...

2. The Connection & Cursor

import sqlite3
from contextlib import closing

# The basic I/O function pattern
def with_db(path: str, fn):
    """Execute fn(cursor) inside a transaction, then commit."""
    with closing(sqlite3.connect(path)) as conn:
        conn.execute("PRAGMA foreign_keys = ON")  # Always enable FKs
        cursor = conn.cursor()
        result = fn(cursor)
        conn.commit()
        return result

Critical: Always enable PRAGMA foreign_keys = ON. SQLite disables foreign key enforcement by default for backward compatibility.

3. Row Factory: The Most Important Feature for DOP

By default, sqlite3 returns rows as tuples. This is fragile—you access data by index.

Option A: sqlite3.Row (Lightweight, Dict-like)

import sqlite3
from contextlib import closing

def fetch_users(db_path: str) -> list[sqlite3.Row]:
    with closing(sqlite3.connect(db_path)) as conn:
        conn.row_factory = sqlite3.Row  # <-- KEY LINE
        cursor = conn.execute("SELECT id, name, email FROM users")
        rows = cursor.fetchall()
        return rows

# Usage: rows behave like dicts AND tuples
for row in rows:
    print(row["name"])      # Dict-like access
    print(row[1])           # Tuple index access
    print(dict(row))        # Convert to plain dict when needed
    print(tuple(row))       # Convert to tuple

Why sqlite3.Row?

  • Memory efficient (stores data once, not copied into dict)
  • Supports key lookup by column name
  • Immutable (good for DOP)
  • Easy to convert: dict(row), tuple(row)

Option B: Dict Factory (True Dicts)

def dict_factory(cursor, row):
    """Return rows as plain dictionaries."""
    fields = [column[0] for column in cursor.description]
    return dict(zip(fields, row))

# Usage
conn.row_factory = dict_factory

When to use which?

  • Use sqlite3.Row for most cases (fast, lightweight)
  • Use dict_factory only if you need to mutate the row or serialize to JSON immediately

4. Parameterized Queries (NEVER Concatenate SQL)

# NEVER DO THIS
user_input = "Robert'); DROP TABLE users; --"
cursor.execute(f"SELECT * FROM users WHERE name = '{user_input}'")  # SQL INJECTION!

# ALWAYS DO THIS: Use ? placeholders
cursor.execute("SELECT * FROM users WHERE name = ? AND age > ?", (user_input, 18))

# For IN clauses, use multiple placeholders
names = ("Alice", "Bob", "Charlie")
placeholders = ",".join(["?"] * len(names))
cursor.execute(f"SELECT * FROM users WHERE name IN ({placeholders})", names)

# For dynamic columns/ordering, validate against a whitelist
ALLOWED_COLUMNS = {"name", "email", "created_at"}
sort_col = user_input_sort
if sort_col not in ALLOWED_COLUMNS:
    raise ValueError(f"Invalid sort column: {sort_col}")
cursor.execute(f"SELECT * FROM users ORDER BY {sort_col}")

Rule: ? for values. Whitelist validation for identifiers (column/table names).

5. Transactions

SQLite transactions are implicit but dangerous if misunderstood:

# Implicit transaction: DML statements (INSERT/UPDATE/DELETE) auto-start a transaction
# Explicit control is better:

def transfer_funds(conn, from_id, to_id, amount):
    """Use conn as context manager for automatic commit/rollback."""
    with conn:  # Commits on success, ROLLBACK on exception
        conn.execute("UPDATE accounts SET balance = balance - ? WHERE id = ?", (amount, from_id))
        conn.execute("UPDATE accounts SET balance = balance + ? WHERE id = ?", (amount, to_id))
    # Automatically committed here

Key behaviors:

  • sqlite3.connect() doesn't auto-commit by default (unless isolation_level=None)
  • A connection context manager (with conn:) wraps in a transaction
  • DDL (CREATE TABLE, etc.) sometimes auto-commits depending on SQLite version

6. Creating Tables & Schema (Procedural)

def init_schema(conn):
    """Idempotent schema setup. Run it every time on startup."""
    schema = """
    CREATE TABLE IF NOT EXISTS users (
        id      INTEGER PRIMARY KEY AUTOINCREMENT,
        name    TEXT NOT NULL,
        email   TEXT UNIQUE,
        meta    TEXT DEFAULT '{}',  -- Store JSON as TEXT
        created INTEGER DEFAULT (unixepoch())
    ) STRICT;  -- SQLite 3.37+: STRICT prevents type affinity gotchas

    CREATE INDEX IF NOT EXISTS idx_users_email ON users(email);
    """
    conn.executescript(schema)

Notes:

  • Use IF NOT EXISTS for idempotency
  • STRICT table option (SQLite 3.37+) enforces types properly
  • Store JSON in TEXT columns; parse in your functions

7. CRUD as Pure Functions

from typing import Optional
import json

# --- CREATE ---
def insert_user(conn, name: str, email: str, meta: dict | None = None) -> int:
    """Returns the new row id."""
    cursor = conn.execute(
        "INSERT INTO users (name, email, meta) VALUES (?, ?, ?)",
        (name, email, json.dumps(meta or {}))
    )
    return cursor.lastrowid

# --- READ ---
def get_user_by_id(conn, user_id: int) -> dict | None:
    conn.row_factory = dict_factory  # Or sqlite3.Row
    cursor = conn.execute("SELECT * FROM users WHERE id = ?", (user_id,))
    row = cursor.fetchone()
    conn.row_factory = None  # Reset if needed
    if row and isinstance(row, dict) and row.get("meta"):
        row["meta"] = json.loads(row["meta"])
    return row

def list_users(conn, limit: int = 100, offset: int = 0) -> list[dict]:
    conn.row_factory = sqlite3.Row
    cursor = conn.execute(
        "SELECT id, name, email, created FROM users ORDER BY created DESC LIMIT ? OFFSET ?",
        (limit, offset)
    )
    rows = cursor.fetchall()
    return [dict(r) for r in rows]

# --- UPDATE ---
def update_user_email(conn, user_id: int, new_email: str) -> int:
    """Returns number of rows modified."""
    cursor = conn.execute(
        "UPDATE users SET email = ? WHERE id = ?",
        (new_email, user_id)
    )
    return cursor.rowcount

# --- DELETE ---
def delete_user(conn, user_id: int) -> int:
    cursor = conn.execute("DELETE FROM users WHERE id = ?", (user_id,))
    return cursor.rowcount

8. Bulk Operations (executemany)

import csv

def bulk_insert_users(conn, users_data: list[tuple[str, str]]):
    """
    users_data: [("Alice", "alice@example.com"), ("Bob", "bob@example.com"), ...]
    """
    conn.executemany(
        "INSERT INTO users (name, email) VALUES (?, ?)",
        users_data
    )

def load_from_csv(conn, filepath: str):
    with open(filepath, newline="") as f:
        reader = csv.reader(f)
        next(reader, None)  # Skip header
        conn.executemany("INSERT INTO users (name, email) VALUES (?, ?)", reader)

Performance tip: Wrap bulk inserts in a transaction (the with conn: block). SQLite can do 50k+ inserts/second in a transaction vs. ~50/second in auto-commit.

9. Working with JSON (DOP Friendly)

SQLite has JSON support (enable with json1 extension, usually built-in):

def find_user_by_meta_key(conn, key: str, value: str) -> list[dict]:
    """Query inside JSON column using SQLite json_extract."""
    conn.row_factory = sqlite3.Row
    cursor = conn.execute(
        "SELECT * FROM users WHERE json_extract(meta, ?) = ?",
        (f"$.{key}", value)
    )
    return [dict(r) for r in cursor.fetchall()]

# Or extract data in SQL
conn.execute("""
    SELECT
        id,
        name,
        json_extract(meta, '$.department') as dept
    FROM users
    WHERE json_extract(meta, '$.active') = true
""")

10. Type Conversions (SQLite ↔ Python)

SQLite is dynamically typed. Python's sqlite3 maps:

SQLite Python
NULL None
INTEGER int
REAL float
TEXT str
BLOB bytes

Custom adapters/converters (use sparingly):

import datetime

# Store datetime as ISO 8601 text
sqlite3.register_adapter(datetime.datetime, lambda dt: dt.isoformat())
sqlite3.register_converter("timestamp", lambda s: datetime.datetime.fromisoformat(s.decode()))

# Connect with detect_types to enable converters
conn = sqlite3.connect(":memory:", detect_types=sqlite3.PARSE_DECLTYPES)

DOP preference: Don't use adapters. Store dates as INTEGER (unixepoch) or TEXT (ISO 8601), and convert in your functions explicitly. Less magic, more transparency.

11. Advanced: Custom Functions in SQL

Register Python functions to use inside SQL queries:

import re

def regexp(pattern: str, text: str) -> bool:
    return re.search(pattern, text) is not None

conn = sqlite3.connect(":memory:")
conn.create_function("REGEXP", 2, regexp)

# Now use in SQL
cursor = conn.execute("SELECT * FROM users WHERE name REGEXP ?", (r"^A.*",))

12. The Complete Pattern: A Small Module

"""users_db.py — Pure procedural SQLite module."""
import sqlite3
from contextlib import closing
from typing import Optional

DB_PATH = "app.db"

# --- Schema ---
def init_db(path: str) -> None:
    with closing(sqlite3.connect(path)) as conn:
        conn.executescript("""
            CREATE TABLE IF NOT EXISTS users (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                email TEXT UNIQUE,
                score REAL DEFAULT 0.0
            ) STRICT;
        """)
        conn.commit()

# --- Queries ---
def add_user(path: str, name: str, email: str) -> int:
    with closing(sqlite3.connect(path)) as conn:
        cursor = conn.execute(
            "INSERT INTO users (name, email) VALUES (?, ?)",
            (name, email)
        )
        conn.commit()
        return cursor.lastrowid

def get_leaderboard(path: str, top_n: int = 10) -> list[sqlite3.Row]:
    with closing(sqlite3.connect(path)) as conn:
        conn.row_factory = sqlite3.Row
        cursor = conn.execute(
            "SELECT name, score FROM users ORDER BY score DESC LIMIT ?",
            (top_n,)
        )
        return cursor.fetchall()

# --- Usage ---
if __name__ == "__main__":
    init_db(DB_PATH)
    uid = add_user(DB_PATH, "Alice", "alice@example.com")
    print(f"Created user {uid}")
    for row in get_leaderboard(DB_PATH):
        print(f"{row['name']}: {row['score']}")

13. Critical Gotchas

  1. SQLite is single-writer: Only one transaction can write at a time. Others wait or get BUSY errors. Use timeout parameter: sqlite3.connect(path, timeout=30.0)

  2. WAL mode for concurrency:

python conn.execute("PRAGMA journal_mode=WAL")

This lets readers proceed while a writer is active.

  1. Bytestrings vs strings: sqlite3 returns TEXT as str in Python 3. In Python 2, this was a mess. You're using Python 3, so no issue.

  2. NULL handling: cursor.fetchone() returns None if no row found. Check before accessing!

  3. Rowcount is unreliable for SELECT in some SQLite versions. Don't depend on it for reads.

14. Logging (Procedural)

El módulo logging usa objetos por dentro, pero tú lo usas de forma totalmente procedural: llamas funciones, pasas datos, listo.

Lo mínimo esencial

import logging

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s %(levelname)s %(message)s"
)

logging.info("App iniciada")
logging.warning("Falta el archivo de config, usando defaults")
logging.error("Fallo al escribir en la base de datos")

Niveles estándar, de menos a más grave:

Nivel Cuándo usarlo
DEBUG Detalles para diagnosticar (valores de variables, SQL ejecutado)
INFO Eventos normales ("usuario creado", "migración completada")
WARNING Algo raro pero recuperable ("reintento 2 de 3")
ERROR Una operación falló
CRITICAL El programa no puede continuar

Nunca uses f-strings en los mensajes

El módulo hace el formateo solo si el mensaje llega a emitirse. Con f-strings pagas el costo siempre:

# MAL: la string se construye aunque DEBUG esté desactivado
logging.debug(f"Filas procesadas: {len(rows)}")

# BIEN: los args se pasan sin tocar, se formatean solo si se emite
logging.debug("Filas procesadas: %d", len(rows))

Configuración procedural reutilizable

import logging
import sys

def setup_logging(level: str = "INFO", log_file: str | None = None) -> None:
    """Configura el logging una vez al inicio. Sin clases, solo datos."""
    fmt = "%(asctime)s | %(levelname)-8s | %(name)s | %(message)s"
    handlers: list[logging.Handler] = [logging.StreamHandler(sys.stderr)]

    if log_file:
        handlers.append(logging.FileHandler(log_file, encoding="utf-8"))

    logging.basicConfig(
        level=getattr(logging, level.upper()),
        format=fmt,
        handlers=handlers,
    )

En el punto de entrada:

def main() -> int:
    setup_logging(level="INFO", log_file="app.log")
    init_db(DB_PATH)
    logging.info("Base de datos inicializada en %s", DB_PATH)
    ...

Loggers por módulo (nombres, no clases)

En vez del logger raíz, cada módulo obtiene el suyo con su nombre — así sabes de dónde viene cada mensaje:

# users_db.py
import logging
log = logging.getLogger(__name__)   # se llamará "users_db"

def add_user(conn, name: str, email: str) -> int:
    log.debug("INSERT user name=%s email=%s", name, email)
    cursor = conn.execute("INSERT INTO users (name, email) VALUES (?, ?)", (name, email))
    conn.commit()
    log.info("Usuario creado id=%d name=%s", cursor.lastrowid, name)
    return cursor.lastrowid

Salida:

2025-01-15 10:32:01 | INFO     | users_db | Usuario creado id=42 name=Alice

Loguear excepciones: exc_info o exception()

Dentro de un except, esto incluye el traceback completo automáticamente:

def transfer(conn, from_id: int, to_id: int, amount: float) -> bool:
    try:
        with conn:
            conn.execute("UPDATE accounts SET balance = balance - ? WHERE id = ?", (amount, from_id))
            conn.execute("UPDATE accounts SET balance = balance + ? WHERE id = ?", (amount, to_id))
        return True
    except sqlite3.IntegrityError:
        log.exception("Transferencia fallida de %d a %d", from_id, to_id)  # loguea el traceback
        return False

Reglas de oro (especialmente con SQLite)

  1. Nunca loguees secrets: passwords, tokens, datos personales. Loguea el id, no el contenido.
  2. Loguea los parámetros SQL, no concatenes SQL: log.debug("query=%s params=%s", sql, params) — nunca log.debug("SELECT ... WHERE name = '%s'" % name).
  3. DEBUG en desarrollo, INFO en producción, con variable de entorno:

python import os setup_logging(level=os.environ.get("LOG_LEVEL", "INFO"))

  1. Los logs no son tu base de datos: para datos que necesitas consultar, usa una tabla SQLite (CREATE TABLE events (...)). Los logs son para humanos leyendo texto.
  2. El logger es thread-safe por diseño: varios hilos pueden escribir sin corromper las líneas.

El patrón completo procedural

"""app.py — punto de entrada"""
import logging
import sys

import users_db

log = logging.getLogger(__name__)

def main() -> int:
    users_db.setup_logging(level="INFO", log_file="app.log")
    log.info("Inicio de la app, args=%s", sys.argv[1:])

    try:
        users_db.init_db(users_db.DB_PATH)
        uid = users_db.add_user(users_db.DB_PATH, "Alice", "alice@example.com")
        log.info("Flujo completado, user_id=%d", uid)
        return 0
    except Exception:
        log.exception("Error fatal, saliendo")
        return 1

if __name__ == "__main__":
    sys.exit(main())

Resumen: setup_logging() una vez al inicio → log = logging.getLogger(__name__) en cada módulo → log.info/debug/warning/error("... %s", dato) en los puntos clave.
